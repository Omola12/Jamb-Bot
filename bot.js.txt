// Store user exam sessions
const userSessions = {};

async function startBot() {
    const { state, saveCreds } = await useMultiFileAuthState('auth_info');
    
    const sock = makeWASocket({
        auth: state,
        printQRInTerminal: false,
    });

    sock.ev.on('connection.update', (update) => {
        const { connection, lastDisconnect, qr } = update;
        
        if (qr) {
            console.log('📱 SCAN THIS QR CODE WITH WHATSAPP:');
            QRCode.generate(qr, { small: true });
        }
        
        if (connection === 'open') {
            console.log('✅ JAMB Exam Bot is LIVE!');
            console.log('📱 Send /start on WhatsApp to begin');
        }
        
        if (connection === 'close') {
            const shouldReconnect = lastDisconnect?.error?.output?.statusCode !== DisconnectReason.loggedOut;
            if (shouldReconnect) {
                console.log('🔄 Reconnecting...');
                startBot();
            }
        }
    });

    sock.ev.on('creds.update', saveCreds);

    sock.ev.on('messages.upsert', async ({ messages }) => {
        const msg = messages[0];
        if (!msg.message || msg.key.fromMe) return;
        
        const sender = msg.key.remoteJid;
        const text = msg.message.conversation || msg.message.extendedTextMessage?.text || '';
        
        if (!text) return;
        
        const lowerText = text.toLowerCase().trim();
        
        // Commands
        if (lowerText === '/start' || lowerText === 'start' || lowerText === 'menu') {
            await sock.sendMessage(sender, { text: `
📚 *JAMB 2026 MOCK EXAM BOT*

*Commands:*
/start - Show this menu
/quiz - Start exam (100 questions)
/score - Check your last score
/help - How to answer

*Exam Info:*
Questions: 100 (JAMB Repeated)
Pass mark: 70%

*How to answer:*
Just send A, B, C, or D for each question

Good luck! 🎯
            `});
            return;
        }
        
        if (lowerText === '/quiz') {
            userSessions[sender] = {
                started: true,
                answers: {},
                currentQuestion: 0,
                score: null,
                percentage: null
            };
            
            const q = questions[0];
            await sock.sendMessage(sender, { text: `
📝 *QUESTION 1/${questions.length}*

${q.text}

A. ${q.A}
B. ${q.B}
C. ${q.C}
D. ${q.D}

*Reply with A, B, C, or D*
            `});
            return;
        }
        
        if (lowerText === '/score') {
            const session = userSessions[sender];
            if (!session || session.score === null) {
                await sock.sendMessage(sender, { text: "📭 You haven't taken the exam yet. Send /quiz to start!" });
            } else {
                await sock.sendMessage(sender, { text: `
📊 *YOUR LAST SCORE*

Score: ${session.score}/${questions.length}
Percentage: ${session.percentage}%
Status: ${session.percentage >= 70 ? '✅ PASSED' : '❌ FAILED'}

Send /quiz to try again!
                `});
            }
            return;
        }
        
        if (lowerText === '/help') {
            await sock.sendMessage(sender, { text: `
📖 *How to Use This Bot*

1. Send /quiz to start the exam
2. For each question, reply with:
   - A (for Option A)
   - B (for Option B)
   - C (for Option C)
   - D (for Option D)

3. After answering all questions, you'll get your score instantly

*Example:* If you think answer is B, just type B and send

Send /quiz to begin!
            `});
            return;
        }
        
        // Handle answers during exam
        const session = userSessions[sender];
        if (session && session.started) {
            const answer = text.toUpperCase().trim();
            
            if (answer === 'A' || answer === 'B' || answer === 'C' || answer === 'D') {
                const currentQ = session.currentQuestion;
                session.answers[currentQ] = answer;
                session.currentQuestion++;
                
                if (session.currentQuestion >= questions.length) {
                    // Calculate score
                    let score = 0;
                    for (let i = 0; i < questions.length; i++) {
                        if (session.answers[i] === questions[i].correct) {
                            score++;
                        }
                    }
                    const percentage = Math.round((score / questions.length) * 100);
                    session.score = score;
                    session.percentage = percentage;
                    session.started = false;
                    
                    // Build result message
                    let resultText = `✅ *EXAM COMPLETE!* ✅\n\n`;
                    resultText += `📊 *Score:* ${score}/${questions.length}\n`;
                    resultText += `📈 *Percentage:* ${percentage}%\n`;
                    resultText += `🎯 *Status:* ${percentage >= 70 ? 'PASSED! 🎉' : 'Failed. Keep practicing! 💪'}\n\n`;
                    resultText += `*Answer Summary (First 15):*\n`;
                    
                    for (let i = 0; i < Math.min(15, questions.length); i++) {
                        const userAns = session.answers[i] || '?';
                        const correct = questions[i].correct;
                        const mark = userAns === correct ? '✅' : '❌';
                        resultText += `${mark} Q${i+1}: You: ${userAns} | Correct: ${correct}\n`;
                    }
                    
                    if (questions.length > 15) {
                        resultText += `\n... and ${questions.length - 15} more questions. Send /quiz to retake full exam!`;
                    }
                    
                    await sock.sendMessage(sender, { text: resultText });
                } else {
                    const q = questions[session.currentQuestion];
                    await sock.sendMessage(sender, { text: `
📝 *QUESTION ${session.currentQuestion + 1}/${questions.length}*

${q.text}

A. ${q.A}
B. ${q.B}
C. ${q.C}
D. ${q.D}

*Reply with A, B, C, or D*
                    `});
                }
            } else {
                await sock.sendMessage(sender, { text: "❌ Invalid! Send A, B, C, or D only." });
            }
        }
    });
}

console.log('🤖 JAMB WhatsApp Exam Bot Starting...');
console.log('📱 Make sure you have WhatsApp on your phone');
console.log('');

startBot();
